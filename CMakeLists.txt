cmake_minimum_required(VERSION 3.24)
project(web_search_tokenizer VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(TOKENIZER_WITH_MONGO "Build MongoDB command line application" ON)
find_package(ICU 67 REQUIRED COMPONENTS uc)
add_library(tokenizer_core STATIC tokenizer/src/tokenizer.cpp tokenizer/src/unicode.cpp
    tokenizer/src/statistics.cpp)
target_include_directories(tokenizer_core PUBLIC tokenizer/include PRIVATE tokenizer/src)
target_link_libraries(tokenizer_core PUBLIC ICU::uc)
target_compile_options(tokenizer_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(tokenizer_tests tokenizer/tests/tokenizer_tests.cpp)
    target_link_libraries(tokenizer_tests PRIVATE tokenizer_core)
    add_test(NAME tokenizer_core COMMAND tokenizer_tests)
endif()
if(TOKENIZER_WITH_MONGO)
    list(APPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/.deps/mongo")
    find_package(mongocxx CONFIG QUIET)
    find_package(bsoncxx CONFIG QUIET)
    if(NOT TARGET mongo::mongocxx_shared OR NOT TARGET mongo::bsoncxx_shared)
        include(cmake/MongoDriver.cmake)
        tokenizer_fetch_mongo()
    endif()
    add_executable(tokenizer tokenizer/src/main.cpp tokenizer/src/repository.cpp
        tokenizer/src/report.cpp)
    target_include_directories(tokenizer PRIVATE tokenizer/src)
    target_link_libraries(tokenizer PRIVATE tokenizer_core mongo::mongocxx_shared mongo::bsoncxx_shared)
    target_compile_options(tokenizer PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(tokenizer PRIVATE TOKENIZER_BUILD_TYPE="$<CONFIG>")
    install(TARGETS tokenizer RUNTIME DESTINATION bin)
endif()
