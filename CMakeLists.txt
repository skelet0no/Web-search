cmake_minimum_required(VERSION 3.24)
project(web_search_tokenizer VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(TOKENIZER_WITH_MONGO "Build MongoDB command line application" ON)
option(ZIPF_BUILD "Build C++ Zipf analysis application (requires Cairo)" OFF)
option(LEMMATIZER_BUILD "Build C++ lemmatizer core" ON)
option(LEMMATIZER_WITH_MONGO "Build lemmatizer MongoDB application" ${TOKENIZER_WITH_MONGO})
option(BOOLEAN_BUILD "Build Boolean index core and tests" OFF)
option(BOOLEAN_BUILD_INDEX "Build MongoDB to file Boolean index application" ${BOOLEAN_BUILD})
option(BOOLEAN_BUILD_SEARCH "Build Boolean search CLI (requires lemmatizer core)" ${BOOLEAN_BUILD})
if(BOOLEAN_BUILD_INDEX OR BOOLEAN_BUILD_SEARCH)
    set(BOOLEAN_BUILD ON)
endif()
if(BOOLEAN_BUILD_SEARCH AND NOT LEMMATIZER_BUILD)
    message(FATAL_ERROR "BOOLEAN_BUILD_SEARCH requires LEMMATIZER_BUILD=ON")
endif()
find_package(ICU 67 REQUIRED COMPONENTS uc)
set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
add_library(text_unicode STATIC text/src/unicode.cpp)
target_include_directories(text_unicode PUBLIC text/include)
target_link_libraries(text_unicode PUBLIC ICU::uc)
add_library(tokenizer_core STATIC tokenizer/src/tokenizer.cpp tokenizer/src/statistics.cpp)
target_include_directories(tokenizer_core PUBLIC tokenizer/include)
target_link_libraries(tokenizer_core PUBLIC text_unicode)
target_compile_options(tokenizer_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(tokenizer_tests tokenizer/tests/tokenizer_tests.cpp)
    target_link_libraries(tokenizer_tests PRIVATE tokenizer_core)
    add_test(NAME tokenizer_core COMMAND tokenizer_tests)
endif()
if(LEMMATIZER_BUILD)
    find_package(OpenSSL REQUIRED)
    include(cmake/XMorphy.cmake)
    add_library(lemmatizer_core STATIC lemmatizer/src/analyzer.cpp
        lemmatizer/src/hash.cpp lemmatizer/src/xmorphy_adapter.cpp)
    target_include_directories(lemmatizer_core PUBLIC lemmatizer/include)
    target_link_libraries(lemmatizer_core PUBLIC text_unicode PRIVATE xmorphy_dictionary OpenSSL::Crypto)
    target_compile_options(lemmatizer_core PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_TESTING)
        add_executable(lemmatizer_tests lemmatizer/tests/lemmatizer_tests.cpp)
        target_link_libraries(lemmatizer_tests PRIVATE lemmatizer_core tokenizer_core)
        add_test(NAME lemmatizer_core COMMAND lemmatizer_tests)
        add_test(NAME lemmatizer_dictionary COMMAND lemmatizer_tests "${LEMMATIZER_DICTIONARY_DIR}")
    endif()
endif()
if(TOKENIZER_WITH_MONGO OR ZIPF_BUILD OR BOOLEAN_BUILD_INDEX OR (LEMMATIZER_BUILD AND LEMMATIZER_WITH_MONGO))
    list(APPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/.deps/mongo")
    find_package(mongocxx 3.11 CONFIG QUIET)
    find_package(bsoncxx 3.11 CONFIG QUIET)
    if(NOT TARGET mongo::mongocxx_shared OR NOT TARGET mongo::bsoncxx_shared)
        include(cmake/MongoDriver.cmake)
        tokenizer_fetch_mongo()
    endif()
endif()
if(BOOLEAN_BUILD)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "Boolean index v1 requires Linux (fsync and atomic renameat2 publication)")
    endif()
    find_package(OpenSSL REQUIRED)
    add_library(boolean_core STATIC boolean/src/index.cpp boolean/src/query.cpp)
    target_include_directories(boolean_core PUBLIC boolean/include)
    target_link_libraries(boolean_core PUBLIC text_unicode OpenSSL::Crypto)
    target_compile_options(boolean_core PRIVATE -Wall -Wextra -Wpedantic)
    if(BOOLEAN_BUILD_INDEX)
        add_library(boolean_repository STATIC boolean/src/repository.cpp)
        target_include_directories(boolean_repository PUBLIC boolean/src PRIVATE lemmatizer/include tokenizer/include)
        target_link_libraries(boolean_repository PUBLIC boolean_core mongo::mongocxx_shared mongo::bsoncxx_shared)
        target_compile_options(boolean_repository PRIVATE -Wall -Wextra -Wpedantic)
        add_executable(boolean-index boolean/src/index_main.cpp)
        target_link_libraries(boolean-index PRIVATE boolean_repository)
        install(TARGETS boolean-index RUNTIME DESTINATION bin)
    endif()
    if(BOOLEAN_BUILD_SEARCH)
        add_library(boolean_language STATIC boolean/src/normalization.cpp)
        target_link_libraries(boolean_language PUBLIC boolean_core tokenizer_core lemmatizer_core)
        target_compile_options(boolean_language PRIVATE -Wall -Wextra -Wpedantic)
        add_executable(boolean-search boolean/src/search_main.cpp)
        target_link_libraries(boolean-search PRIVATE boolean_language)
        target_compile_options(boolean-search PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_definitions(boolean-search PRIVATE BOOLEAN_DEFAULT_DICTIONARY="${LEMMATIZER_DICTIONARY_DIR}")
        install(TARGETS boolean-search RUNTIME DESTINATION bin)
    endif()
    if(BUILD_TESTING)
        add_executable(boolean_tests boolean/tests/core_tests.cpp)
        target_link_libraries(boolean_tests PRIVATE boolean_core tokenizer_core)
        add_test(NAME boolean_core COMMAND boolean_tests)
        if(BOOLEAN_BUILD_SEARCH)
            add_executable(boolean_morphology_tests boolean/tests/morphology_tests.cpp)
            target_link_libraries(boolean_morphology_tests PRIVATE boolean_language)
            add_test(NAME boolean_morphology COMMAND boolean_morphology_tests "${LEMMATIZER_DICTIONARY_DIR}")
            add_test(NAME boolean_search_help COMMAND boolean-search --help)
        endif()
        if(BOOLEAN_BUILD_INDEX)
            add_test(NAME boolean_index_help COMMAND boolean-index --help)
            add_executable(boolean_repository_tests boolean/tests/repository_tests.cpp)
            target_link_libraries(boolean_repository_tests PRIVATE boolean_repository tokenizer_core)
            target_include_directories(boolean_repository_tests PRIVATE lemmatizer/include)
            add_test(NAME boolean_repository COMMAND boolean_repository_tests)
            set_tests_properties(boolean_repository PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180)
        endif()
    endif()
endif()
if(ZIPF_BUILD)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo)
    add_library(zipf_core STATIC zipf/src/analysis.cpp zipf/src/report.cpp)
    target_include_directories(zipf_core PUBLIC zipf/src)
    target_link_libraries(zipf_core PUBLIC PkgConfig::CAIRO)
    target_compile_options(zipf_core PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(zipf zipf/src/main.cpp)
    target_link_libraries(zipf PRIVATE zipf_core mongo::mongocxx_shared mongo::bsoncxx_shared)
    target_compile_options(zipf PRIVATE -Wall -Wextra -Wpedantic)
    install(TARGETS zipf RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
        add_executable(zipf_tests zipf/tests/analysis_tests.cpp)
        target_link_libraries(zipf_tests PRIVATE zipf_core)
        add_test(NAME zipf_core COMMAND zipf_tests)
        add_executable(zipf_input_tests zipf/tests/input_tests.cpp)
        target_include_directories(zipf_input_tests PRIVATE zipf/src)
        target_link_libraries(zipf_input_tests PRIVATE mongo::bsoncxx_shared)
        add_test(NAME zipf_input COMMAND zipf_input_tests)
    endif()
endif()
if(TOKENIZER_WITH_MONGO)
    add_executable(tokenizer tokenizer/src/main.cpp tokenizer/src/repository.cpp
        tokenizer/src/report.cpp)
    target_include_directories(tokenizer PRIVATE tokenizer/src)
    target_link_libraries(tokenizer PRIVATE tokenizer_core mongo::mongocxx_shared mongo::bsoncxx_shared)
    target_compile_options(tokenizer PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(tokenizer PRIVATE TOKENIZER_BUILD_TYPE="$<CONFIG>")
    install(TARGETS tokenizer RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
        add_executable(tokenizer_repository_tests tokenizer/tests/repository_tests.cpp)
        target_link_libraries(tokenizer_repository_tests PRIVATE mongo::mongocxx_shared mongo::bsoncxx_shared)
        add_dependencies(tokenizer_repository_tests tokenizer)
        add_test(NAME tokenizer_repository COMMAND tokenizer_repository_tests "$<TARGET_FILE:tokenizer>")
        set_tests_properties(tokenizer_repository PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120)
    endif()
endif()
if(LEMMATIZER_BUILD AND LEMMATIZER_WITH_MONGO)
    add_library(lemmatizer_application STATIC lemmatizer/src/options.cpp
        lemmatizer/src/repository.cpp lemmatizer/src/statistics.cpp lemmatizer/src/report.cpp)
    target_include_directories(lemmatizer_application PUBLIC lemmatizer/src)
    target_link_libraries(lemmatizer_application PUBLIC lemmatizer_core mongo::mongocxx_shared mongo::bsoncxx_shared)
    target_compile_options(lemmatizer_application PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(lemmatizer_application PRIVATE LEMMATIZER_BUILD_TYPE="$<CONFIG>"
        LEMMATIZER_DEFAULT_DICTIONARY="${LEMMATIZER_DICTIONARY_DIR}")
    add_executable(lemmatizer lemmatizer/src/main.cpp)
    target_link_libraries(lemmatizer PRIVATE lemmatizer_application)
    install(TARGETS lemmatizer RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
        add_executable(lemmatizer_repository_tests lemmatizer/tests/repository_tests.cpp)
        target_link_libraries(lemmatizer_repository_tests PRIVATE lemmatizer_application tokenizer_core)
        add_test(NAME lemmatizer_repository COMMAND lemmatizer_repository_tests "${LEMMATIZER_DICTIONARY_DIR}")
        set_tests_properties(lemmatizer_repository PROPERTIES SKIP_RETURN_CODE 77)
    endif()
endif()
